\setotherlanguage{english}

% --- Fonts -------------------------------------------------------------------

% --- Floats, tables, units, code ---------------------------------------------

\usepackage{csquotes}
% csquotes has no Turkish style; Turkish uses “...” and ‘...’.

\addbibresource{references.bib}

% --- Appendices --------------------------------------------------------------
\usepackage[titletoc,title]{appendix}

% --- Links, references, acronyms (hyperref late, cleveref last) --------------
\usepackage[hidelinks]{hyperref}
% cleveref has no Turkish support and mis-detects polyglossia languages:
% pin it to English and define Turkish names below.
\usepackage[english,nameinlink,noabbrev]{cleveref}
\usepackage[printonlyused]{acronym}

% cleveref has no Turkish language file: define the names explicitly.
\crefname{chapter}{Bölüm}{Bölümler}

\Crefname{section}{Kısım}{Kısımlar}
\crefname{subsection}{Kısım}{Kısımlar}
\Crefname{subsection}{Kısım}{Kısımlar}
\crefname{appendix}{Ek}{Ekler}
\Crefname{appendix}{Ek}{Ekler}
\crefname{figure}{Şekil}{Şekiller}
\Crefname{figure}{Şekil}{Şekiller}
\crefname{table}{Tablo}{Tablolar}
\Crefname{table}{Tablo}{Tablolar}
\crefname{equation}{Denklem}{Denklemler}
\Crefname{equation}{Denklem}{Denklemler}
\crefname{theorem}{Teorem}{Teoremler}
\Crefname{theorem}{Teorem}{Teoremler}
\creflabelformat{equation}{#2(#1)#3}

\newtheorem{theorem}{Teorem}[chapter]
